control.pause=KeyP | Pause
control.undo=KeyU | Undo (where allowed)

control.camera=KeyC | Reset / cycle camera
control.mute=KeyM | Mute
control.board=KeyB | Toggle HTML board
control.help=Slash | Help
control.skip=KeyF | Skip animation
